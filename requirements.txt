# Pinned dependencies for a reproducible OFFLINE install.
# Build a local wheel cache on a machine WITH internet, then install from it on
# the offline target:
#
#   (online build box)  pip download -r requirements.txt -d wheelhouse
#   (offline target)    pip install --no-index --find-links wheelhouse -r requirements.txt
#
# graphrag is pinned exactly: the LLM model-registry and settings schema are
# version sensitive and this project is coded against 3.1.0.
graphrag==3.1.0

# PDF -> Markdown with high-fidelity (nested / multi-page) table reconstruction.
# Docling downloads ML models on first use; they MUST be pre-cached offline.
# See scripts/prefetch_docling_models.py.
docling>=2.7.0,<3.0

# Local on-disk HuggingFace embedding model (safetensors) loaded offline.
sentence-transformers>=3.0.0,<6.0

# LangChain query surface (BaseRetriever + @tool). The full `langchain` package
# is intentionally NOT required — only these two are used. lancedb comes in
# transitively via graphrag (graphrag-vectors).
langchain-core>=0.3.0
langchain-openai>=0.2.0

# Storage / data (pandas+pyarrow are used directly for parquet/manifest IO).
pandas>=2.0.0
pyarrow>=15.0.0

# CLI + config
typer>=0.12.0
PyYAML>=6.0
python-dotenv>=1.0.0
